Animator::Animator(GameObject& associated) 
    : Component(associated), frameStart(0), frameEnd(0), frameTime(0.0f), currentFrame(0), timeElapsed(0.0f) {
}

void Animator::Update(float dt) {
    //se frameTime for 0, não tem animação
    if (frameTime == 0.0f) {
        return; 
    }

    timeElapsed += dt; //mudar depois pelo delta time de vdd

    //passou mais tempo que o tempo do frame
    if (timeElapsed > frameTime) {
        currentFrame++;
        timeElapsed -= frameTime;

        //passou do ultimo quadro
        if (currentFrame > frameEnd) {
            currentFrame = frameStart;
        }

        //atualiza o frame no SpriteRenderer do mesmo objeto
        SpriteRenderer* sprite = associated.GetComponent<SpriteRenderer>();
        if (sprite != nullptr) {
            sprite->SetFrame(currentFrame);
        }
    }
}

void Animator::Render() {
}

void Animator::SetAnimation(std::string name) {
    //avoid restarting the same animation if already active
    if (currentAnimationName == name) {
        return;
    }

    auto it = animations.find(name);
    if (it != animations.end()) {
        currentAnimationName = name;
        frameStart = it->second.frameStart;
        frameEnd = it->second.frameEnd;
        frameTime = it->second.frameTime;

        currentFrame = frameStart;
        timeElapsed = 0.0f;

        //chama SetFrame no SpriteRenderer
        SpriteRenderer* sprite = associated.GetComponent<SpriteRenderer>();
        if (sprite != nullptr) {
            sprite->SetFrame(currentFrame);
        }
    }
}

void Animator::AddAnimation(std::string name, Animation anim) {
    //insere o par se não encontrar a animação no mapa
    if (animations.find(name) == animations.end()) {
        animations[name] = anim; 
    }
}

Bullet::Bullet(GameObject& associated, float angle, float speed, int damage, float maxDistance)
    : Component(associated), distanceLeft(maxDistance), damage(damage) {

    SpriteRenderer* renderer = new SpriteRenderer(associated, "assets/img/Bullet.png");
    associated.AddComponent(renderer);

    // calcula velocidade a partir do angulo
    float rad = angle * 3.14159265f / 180.0f;
    this->speed = Vec2(std::cos(rad) * speed, std::sin(rad) * speed);

    associated.box.w = 8;
    associated.box.h = 8;
}

void Bullet::Update(float dt) {
    associated.box.x += speed.x * dt;
    associated.box.y += speed.y * dt;
    distanceLeft -= this->speed.Magnitude() * dt;

    if (distanceLeft <= 0) {
        associated.RequestDelete();
    }
}

void Bullet::Render() {}

int Bullet::GetDamage() {
    return damage;
}

void Camera::Update(float dt){
    if(focus != nullptr) {
        pos.x = focus->box.Center().x - 600.0f;
        pos.y = focus->box.Center().y - 450.0f;
    } 
    else{
        InputManager& input = InputManager::GetInstance();
        
        float camSpeed = 300.0f; 
        
        speed = Vec2(0, 0);

        if(input.IsKeyDown(LEFT_ARROW_KEY)){
            speed.x = -camSpeed;
        }
        if(input.IsKeyDown(RIGHT_ARROW_KEY)){
            speed.x = camSpeed;
        }
        if(input.IsKeyDown(UP_ARROW_KEY)){
            speed.y = -camSpeed;
        }
        if(input.IsKeyDown(DOWN_ARROW_KEY)){
            speed.y = camSpeed;
        }

        pos.x += speed.x * dt;
        pos.y += speed.y * dt;
    }
}

Character* Character::player = nullptr;

Character::Character(GameObject& associated, std::string sprite)
    : Component(associated), speed(0, 0), linearSpeed(200.0f), hp(100) {
    // sprite do personagem
    SpriteRenderer* renderer = new SpriteRenderer(associated, sprite, 1, 4);
    associated.AddComponent(renderer);

    // animações
    Animator* animator = new Animator(associated);
    animator->AddAnimation("idle", Animation(0, 0, 0.0f));
    animator->AddAnimation("walking", Animation(0, 3, 0.1f));
    animator->AddAnimation("dead", Animation(0, 0, 0.0f));
    associated.AddComponent(animator);
    animator->SetAnimation("idle");

    // registra instância global do jogador
    player = this;
}

Character::~Character() {
    player = nullptr;
}

void Character::Start() {
    // cria a arma como objeto separado no State
    GameObject* gunObj = new GameObject();
    Gun* gunComp = new Gun(*gunObj, std::weak_ptr<GameObject>());
    gunObj->AddComponent(gunComp);
    Game::GetInstance().GetState().AddObject(gunObj);

    // guarda referência fraca para a arma
    std::weak_ptr<GameObject> gunWeak = Game::GetInstance().GetState().GetObjectPtr(gunObj);
    gun = gunWeak;
}

void Character::Update(float dt) {
    if (hp <= 0) {
        // animação de morte
        Animator* animator = associated.GetComponent<Animator>();
        if (animator) {
            animator->SetAnimation("dead");
        }
        deathTimer.Update(dt);
        if (deathTimer.Get() >= 1.0f) {
            associated.RequestDelete();
        }
        return;
    }

    // consome fila de comandos
    if (!taskQueue.empty()) {
        Command cmd = taskQueue.front();
        taskQueue.pop();

        if (cmd.type == Command::MOVE) {
            speed = cmd.pos.GetNormalized() * linearSpeed;

            // orienta animação de movimento
            Animator* animator = associated.GetComponent<Animator>();
            if (animator) {
                animator->SetAnimation("walking");
            }
        } else if (cmd.type == Command::SHOOT) {
            auto gunObj = gun.lock();
            if (gunObj) {
                Gun* gunComp = gunObj->GetComponent<Gun>();
                if (gunComp) {
                    gunComp->Shoot(cmd.pos);
                }
            }

            Animator* animator = associated.GetComponent<Animator>();
            if (animator) {
                animator->SetAnimation("idle");
            }
        }
    }

    // aplica movimento
    associated.box.x += speed.x * dt;
    associated.box.y += speed.y * dt;

    // atrito para desacelerar suavemente
    speed.x *= 0.90f;
    speed.y *= 0.90f;

    // se parou, volta para idle
    if (speed.Magnitude() < 1.0f) {
        Animator* animator = associated.GetComponent<Animator>();
        if (animator) {
            animator->SetAnimation("idle");
        }
    }
}

void Character::Render() {}

void Character::Issue(Command task) {
    taskQueue.push(task);
}

Game::Game(std::string title, int width, int height){
    if(instance != nullptr){
        std::cerr << "Erro na lógica do jogo. Instância de jogo já existe" << std::endl;
        //erro tratar depois
        return;
    }
    instance = this;    

    //core SDL3 subsystems
    if(!SDL_Init(SDL_INIT_VIDEO | SDL_INIT_AUDIO)){
        std::cerr << "SDL_Init Error: " << SDL_GetError() << std::endl;
    }
    //retirei check de load das imagens pq aparentemente removeram as funções
    if (!MIX_Init()) { 
    std::cerr << "MIX_Init Error: " << SDL_GetError() << std::endl;
}
    //SDL3_mixer device
    mixer = MIX_CreateMixerDevice(SDL_AUDIO_DEVICE_DEFAULT_PLAYBACK, nullptr);
    if(mixer == nullptr){
        std::cerr << "Mixer Error: " << SDL_GetError() << std::endl;
    }
    //SDL3 window
    window = SDL_CreateWindow(title.c_str(), width, height, 0);
    if(window == nullptr){
        std::cerr << "SDL_CreateWindow Error: " << SDL_GetError() << std::endl;
    }
    //SDL3 renderer
    renderer = SDL_CreateRenderer(window, nullptr);
    if(renderer == nullptr){
        std::cerr << "SDL_CreateRenderer Error: " << SDL_GetError() << std::endl;
    }

    state = new State();
}

Game::~Game(){
    delete state;
    if(mixer) MIX_DestroyMixer(mixer);
    MIX_Quit();
    //IMG_Quit();
    SDL_DestroyRenderer(renderer);
    SDL_DestroyWindow(window);
    SDL_Quit(); 
}

    return *state;

    return renderer;

void Game::CalculateDeltaTime() {
    Uint64 currentTime = SDL_GetTicks();
    dt = (currentTime - frameStart) / 1000.0f;
    frameStart = currentTime;
}

float Game::GetDeltaTime() {
    return dt;
}

void Game::Run(){
    frameStart = SDL_GetTicks();
    state->Start();
    while (!state->QuitRequested()) {

        CalculateDeltaTime();
        InputManager::GetInstance().Update();
        state->Update(GetDeltaTime());

        state->Render();
        SDL_RenderPresent(renderer);
        SDL_Delay(33); //30 fps
    }
}

GameObject::~GameObject() {
    for (int i = components.size() - 1; i >= 0; i--){
        delete components[i];
    }
    components.clear();

void GameObject::Start() {
    for (int i = 0; i < components.size(); i++) {
        components[i]->Start();
    }
    started = true;
}

void GameObject::Update(float dt) {
    for (int i = 0; i < components.size(); i++){
        components[i]->Update(dt);
    }
}

void GameObject::Render() {
    for (int i = 0; i < components.size(); i++){
        components[i]->Render();
    }
}

    return isDead;
}

    isDead = true;
}

void GameObject::AddComponent(Component* cpt){
    components.emplace_back(cpt);
    if (started) {
        cpt->Start();
    }
}

void GameObject::RemoveComponent(Component* cpt){
    auto it = std::find(components.begin(), components.end(), cpt);
    if (it != components.end()) {
        components.erase(it);
    }
}

Gun::Gun(GameObject& associated, std::weak_ptr<GameObject> character)
    : Component(associated), character(character), cooldownState(0), angle(0.0f) {
    // sprite da arma
    SpriteRenderer* renderer = new SpriteRenderer(associated, "assets/img/Gun.png");
    associated.AddComponent(renderer);

    shotSound.Open("assets/audio/BGM.wav");
    reloadSound.Open("assets/audio/BGM.wav");
}

void Gun::Update(float dt) {
    cdTimer.Update(dt);

    auto charObj = character.lock();
    if (!charObj) {
        associated.RequestDelete();
        return;
    }

    // sincroniza a posição da arma com o personagem
    Vec2 center = charObj->box.Center();
    associated.box.x = center.x - associated.box.w / 2.0f;
    associated.box.y = center.y - associated.box.h / 2.0f;

    // aplica deslocamento na direção do ângulo
    float rad = angle * 3.14159265f / 180.0f;
    associated.box.x += std::cos(rad) * 12.0f;
    associated.box.y += std::sin(rad) * 12.0f;

    // máquina de estados de cooldown (0 a 3)
    if (cooldownState == 1) {
        // disparo efetuado, entrando em recarga
        if (cdTimer.Get() >= 0.2f) {
            cooldownState = 2;
            cdTimer.Restart();
            reloadSound.Play(1);
        }
    } else if (cooldownState == 2) {
        // animação de recarga (pump-action)
        if (cdTimer.Get() >= 0.5f) {
            cooldownState = 3;
            cdTimer.Restart();
        }
    } else if (cooldownState == 3) {
        // fim da recarga, pronto para atirar novamente
        cooldownState = 0;
    }
}

void Gun::Render() {}

void Gun::Shoot(Vec2 target) {
    if (cooldownState != 0) {
        return;
    }

    // calcula ângulo para o alvo
    Vec2 pos = associated.box.Center();
    Vec2 dir = target - pos;
    angle = dir.Inclination() * 180.0f / 3.14159265f;

    shotSound.Play(1);

    // instancia projétil
    GameObject* bulletObj = new GameObject();
    bulletObj->box.x = pos.x - 4;
    bulletObj->box.y = pos.y - 4;

    float speed = 500.0f;
    int damage = 25;
    float maxDistance = 600.0f;

    Bullet* bullet = new Bullet(*bulletObj, angle, speed, damage, maxDistance);
    bulletObj->AddComponent(bullet);

    Game::GetInstance().GetState().AddObject(bulletObj);

    // entra em cooldown
    cooldownState = 1;
    cdTimer.Restart();
}

InputManager::InputManager() : quitRequested(false), updateCounter(0), mouseX(0), mouseY(0) {
    for (int i = 0; i < 6; i++) {
        mouseState[i] = false;
        mouseUpdate[i] = 0;
    } //só tem 5 botoes principais na SDL3
}

InputManager::~InputManager() {
}

void InputManager::Update() {
    SDL_Event event;

    float fX, fY;
    SDL_GetMouseState(&fX, &fY);
    mouseX = static_cast<int>(fX);
    mouseY = static_cast<int>(fY);

    quitRequested = false;
    updateCounter++;

    while (SDL_PollEvent(&event)) {
        if (event.type == SDL_EVENT_QUIT) {
            quitRequested = true;
        }
        else if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN || event.type == SDL_EVENT_MOUSE_BUTTON_UP) {
            int button = event.button.button;
            mouseState[button] = (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN);
            mouseUpdate[button] = updateCounter;
        }

        else if (event.type == SDL_EVENT_KEY_DOWN || event.type == SDL_EVENT_KEY_UP) {
            // ignora o auto-repeat quando segura tecla
            if (!event.key.repeat) {
                // Em SDL3 a tecla é acessada via event.key.key (em SDL2 era event.key.keysym.sym)
                int key = event.key.key;
                
                // unordered_map cria o elemento automaticamente se não existir
                keyState[key] = (event.type == SDL_EVENT_KEY_DOWN);
                keyUpdate[key] = updateCounter;
            }
        }
    }
}

bool InputManager::KeyPress(int key) {
    return keyState[key] && (keyUpdate[key] == updateCounter);
}

bool InputManager::KeyRelease(int key) {
    return !keyState[key] && (keyUpdate[key] == updateCounter);
}

bool InputManager::IsKeyDown(int key) {
    return keyState[key];
}

bool InputManager::MousePress(int button) {
    return mouseState[button] && (mouseUpdate[button] == updateCounter);
}

bool InputManager::MouseRelease(int button) {
    return !mouseState[button] && (mouseUpdate[button] == updateCounter);
}

bool InputManager::IsMouseDown(int button) {
    return mouseState[button];
}

int InputManager::GetMouseX() {
    return mouseX;
}

int InputManager::GetMouseY() {
    return mouseY;
}

bool InputManager::QuitRequested() {
    return quitRequested;
}

PlayerController::PlayerController(GameObject& associated) : Component(associated) {}

void PlayerController::Update(float dt) {
    auto charObj = character.lock();
    if (!charObj) return;

    Character* characterComp = charObj->GetComponent<Character>();
    if (!characterComp) return;

    InputManager& input = InputManager::GetInstance();

    Vec2 moveDir(0, 0);
    if (input.IsKeyDown(SDLK_W) || input.IsKeyDown(UP_ARROW_KEY)) moveDir.y -= 1;
    if (input.IsKeyDown(SDLK_S) || input.IsKeyDown(DOWN_ARROW_KEY)) moveDir.y += 1;
    if (input.IsKeyDown(SDLK_A) || input.IsKeyDown(LEFT_ARROW_KEY)) moveDir.x -= 1;
    if (input.IsKeyDown(SDLK_D) || input.IsKeyDown(RIGHT_ARROW_KEY)) moveDir.x += 1;

    if (moveDir.x != 0 || moveDir.y != 0) {
        characterComp->Issue(Command(Command::MOVE, moveDir.x, moveDir.y));
    }

    if (input.MousePress(LEFT_MOUSE_BUTTON)) {
        Vec2 target(input.GetMouseX(), input.GetMouseY());
        target.x += Camera::pos.x;
        target.y += Camera::pos.y;
        characterComp->Issue(Command(Command::SHOOT, target.x, target.y));
    }
}

std::unordered_map<std::string, SDL_Texture*> Resources::imageTable;
std::unordered_map<std::string, MIX_Audio*> Resources::audioTable;

SDL_Texture* Resources::GetImage(std::string file) {
    auto it = imageTable.find(file);
    
    if (it == imageTable.end()) {
        SDL_Renderer* renderer = Game::GetInstance().GetRenderer();
        SDL_Texture* texture = IMG_LoadTexture(renderer, file.c_str());
        
        if (texture == nullptr) {
            std::cerr << "Erro ao carregar textura: " << SDL_GetError() << std::endl;
            return nullptr;
        }
        
        imageTable[file] = texture;
        return texture;
    }
    
    //se ja existe retorna o ponteiro do cache
    return it->second;
}

MIX_Audio* Resources::GetAudio(std::string file) {
    auto it = audioTable.find(file);

    if (it == audioTable.end()) {
        MIX_Mixer* mixer = Game::GetInstance().GetMixer();
        MIX_Audio* audio = MIX_LoadAudio(mixer, file.c_str(), true);
        
        if (audio == nullptr) {
            std::cerr << "Erro ao carregar audio: " << SDL_GetError() << std::endl;
            return nullptr;
        }
        
        audioTable[file] = audio;
        return audio;
    }
    
    return it->second;
}

void Resources::ClearImages() {
    for (auto it = imageTable.begin(); it != imageTable.end(); it++) {
        if (it->second != nullptr) {
            SDL_DestroyTexture(it->second);
        }
    }
    imageTable.clear();
}

void Resources::ClearAudios() {
    for (auto it = audioTable.begin(); it != audioTable.end(); it++) {
        if (it->second != nullptr) {
            MIX_DestroyAudio(it->second);
        }
    }
    audioTable.clear();
}

void Resources::ClearAll() {
    ClearImages();
    ClearAudios();
}

    clipRect.y = static_cast<float>(y);
    clipRect.w = static_cast<float>(w);
    clipRect.h = static_cast<float>(h);
}

void Sprite::SetCameraFollower(bool follow) {
    cameraFollower = follow;
}

void Sprite::Render(float x, float y, float w, float h, double angle){
    if (texture){
        //subtrai a camera para nao seguir
        float renderX = x - (cameraFollower ? 0 : Camera::pos.x);
        float renderY = y - (cameraFollower ? 0 : Camera::pos.y);

        SDL_FRect dstRect = { renderX, renderY, w, h };
        SDL_RenderTextureRotated(Game::GetInstance().GetRenderer(), texture, &clipRect, &dstRect, angle, nullptr, flip);
    }
}

void Sprite::SetFrameCount(int frameCountW, int frameCountH){
    this->frameCountW = frameCountW;
    this->frameCountH = frameCountH;
}

void Sprite::SetFrame(int frame) {
    int frameW = GetWidth();
    int frameH = GetHeight();
    
    int currentX = (frame % frameCountW) * frameW;
    int currentY = (frame / frameCountW) * frameH;
    
    SetClip(currentX, currentY, frameW, frameH);
}

float Sprite::GetWidth() {
    return (width / frameCountW) * scale.x;
}

float Sprite::GetHeight() {
    return (height / frameCountH) * scale.y;
}

bool Sprite::IsOpen(){
    return texture != nullptr;
}

void Sprite::SetScale(float scaleX, float scaleY) {
    if (scaleX != 0) {
        scale.x = scaleX;
    }
    if (scaleY != 0) {
        scale.y = scaleY;
    }
}

Vec2 Sprite::GetScale() {
    return scale;
}

void Sprite::SetFlip(SDL_FlipMode f) {
    flip = f;
}

    for (int i = 0; i < objectArray.size(); i++) {

        }
    }
    
    return std::weak_ptr<GameObject>();

void State::Update(float dt){
    InputManager& input = InputManager::GetInstance();
    
    if(input.QuitRequested() || input.KeyPress(ESCAPE_KEY)){
        quitRequested = true;
    }

    Camera::Update(dt);

    if(input.KeyPress(SPACE_KEY)){
        GameObject* zombieObj = new GameObject();
        
        zombieObj->box.x = input.GetMouseX() + Camera::pos.x - (zombieObj->box.w / 2.0f);
        zombieObj->box.y = input.GetMouseY() + Camera::pos.y - (zombieObj->box.h / 2.0f);

        Zombie* zombieLogic = new Zombie(*zombieObj);
        zombieObj->AddComponent(zombieLogic);
        AddObject(zombieObj);
    }

    Camera::Update(dt);

    for(int i = 0; i < objectArray.size(); i++){
        objectArray[i]->Update(dt);
    }

    for(int i = 0; i < objectArray.size(); i++){
        if(objectArray[i]->IsDead()){
            objectArray.erase(objectArray.begin() + i);
            i--; 
        }
    }
}

        objectArray[i]->Render();
    }

    : Component(associated) {

void TileMap::Load(std::string file) {
    std::ifstream f(file);
    if (!f.is_open()) {
        std::cerr << "Erro ao abrir o mapa: " << file << std::endl;
        return;
    }

    std::string token;
    
    std::getline(f, token, ',');
    mapWidth = std::stoi(token);
    
    std::getline(f, token, ',');
    mapHeight = std::stoi(token);
    
    std::getline(f, token, ',');
    mapDepth = std::stoi(token);

    int totalTiles = mapWidth * mapHeight * mapDepth;
    
    for (int i = 0; i < totalTiles; i++) {
        std::getline(f, token, ',');
        tileMatrix.push_back(std::stoi(token));
    }
}

void TileMap::SetTileSet(TileSet* tileSet) {
    // O unique_ptr assume a posse do ponteiro usando reset
    this->tileSet.reset(tileSet);
}

int& TileMap::At(int x, int y, int z) {
    // Converte a coordenada 3D num índice de matriz unidimensional
    int index = x + (y * mapWidth) + (z * mapWidth * mapHeight);
    return tileMatrix[index];
}

void TileMap::RenderLayer(int layer) {
    for (int x = 0; x < mapWidth; x++) {
        for (int y = 0; y < mapHeight; y++) {
            
            float posX = x * tileSet->GetTileWidth();
            float posY = y * tileSet->GetTileHeight();
            
            posX += associated.box.x;
            posY += associated.box.y;
            
            int tileIndex = At(x, y, layer);
            
            if (tileIndex > -1) {
                tileSet->RenderTile(tileIndex, posX, posY);
            }
        }
    }
}

void TileMap::Render() {
    for (int z = 0; z < mapDepth; z++) {
        RenderLayer(z);
    }
}

TileSet::TileSet(int tileWidth, int tileHeight, std::string file)
    : tileWidth(tileWidth), tileHeight(tileHeight), tileSet(file){

    if(tileSet.IsOpen()){
        columns = tileSet.GetWidth() / tileWidth;
        rows = tileSet.GetHeight() / tileHeight;
    } else{
        columns = 0;
        rows = 0;
    }
}

void TileSet::RenderTile(unsigned index, float x, float y){
    unsigned maxTiles = rows * columns;
    if(index < maxTiles){
        int tileColumn = index % columns;
        int tileRow = index / columns;

        int clipX = tileColumn * tileWidth;
        int clipY = tileRow * tileHeight;

        tileSet.SetClip(clipX, clipY, tileWidth, tileHeight);
        tileSet.Render(x, y, tileWidth, tileHeight);
    }
}

int TileSet::GetTileWidth(){
    return tileWidth;
}

int TileSet::GetTileHeight(){
    return tileHeight;
}

Vec2::Vec2() : x(0), y(0) {}
Vec2::Vec2(float x, float y) : x(x), y(y) {}

//rotação baseada na matriz de rotação especificada no PDF
Vec2 Vec2::GetRotated(float angle) const{
    float x_prime = x * std::cos(angle) - y * std::sin(angle);
    float y_prime = y * std::cos(angle) + x * std::sin(angle);
    return Vec2(x_prime, y_prime);
}

Vec2 Vec2::operator+(const Vec2& rhs) const { return Vec2(x + rhs.x, y + rhs.y); }
Vec2 Vec2::operator-(const Vec2& rhs) const { return Vec2(x - rhs.x, y - rhs.y); }
Vec2 Vec2::operator*(const float rhs) const { return Vec2(x * rhs, y * rhs); }

float Vec2::Magnitude() const { return std::sqrt(x * x + y * y); }

Vec2 Vec2::GetNormalized() const{
    float mag = Magnitude();
    if (mag == 0) return Vec2(0, 0); //prevenção de divisão por zero
    return Vec2(x / mag, y / mag);
}

float Vec2::Distance(const Vec2& dst) const { return (*this - dst).Magnitude(); }

float Vec2::Inclination() const { return std::atan2(y, x); }

float Vec2::Inclination(const Vec2& dst) const { return (dst - *this).Inclination(); }
